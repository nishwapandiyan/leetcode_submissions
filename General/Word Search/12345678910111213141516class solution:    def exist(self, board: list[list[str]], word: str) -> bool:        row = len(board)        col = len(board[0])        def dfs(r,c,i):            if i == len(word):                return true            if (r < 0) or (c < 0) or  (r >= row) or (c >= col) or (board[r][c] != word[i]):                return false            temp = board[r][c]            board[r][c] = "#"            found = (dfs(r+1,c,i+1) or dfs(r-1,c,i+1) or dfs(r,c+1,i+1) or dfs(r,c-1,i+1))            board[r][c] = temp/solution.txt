class Solution:
    def exist(self, board: List[List[str]], word: str) -> bool:
        row = len(board)
        col = len(board[0])

        def dfs(r,c,i):
            if i == len(word):
                return True
            if (r < 0) or (c < 0) or  (r >= row) or (c >= col) or (board[r][c] != word[i]):
                return False
            temp = board[r][c]
            board[r][c] = "#"

            found = (dfs(r+1,c,i+1) or dfs(r-1,c,i+1) or dfs(r,c+1,i+1) or dfs(r,c-1,i+1))
            board[r][c] = temp
